      <div class="ui-panel" id="uiPanel">
        <h3 id="dialogueText">Loading World Data from Lexicon...</h3>
        <button class="btn-action" id="btnStrike" style="display:none;" onclick="triggerChannelStrike()">Channel Strike (Pay $50)</button>
      </div>
    </div>

    <script>
      // 1. Initialize Game Data
      let currentGameState = null;

      window.onload = function() {
        // Silently fetch the spreadsheet data on load
        google.script.run.withSuccessHandler(initializeGame).getGameData();
      };

      function initializeGame(data) {
        currentGameState = JSON.parse(data);
        
        const dialogue = document.getElementById('dialogueText');
        const btn = document.getElementById('btnStrike');
        
        dialogue.innerHTML = `A wild <b>${currentGameState.lexicon.fact_venmo || 'Vanguard'}</b> blocks the bridge!<br>Boss HP: $${currentGameState.venmoBossHP}`;
        
        if(currentGameState.venmoBossHP > 0) {
          btn.style.display = "inline-block";
        } else {
          dialogue.innerHTML = "The bridge is clear! Whispering Woods unlocked.";
        }
      }

      // 2. Combat Mechanic (Logs payment to Sheet)
      function triggerChannelStrike() {
        const paymentAmount = 50; 
        document.getElementById('dialogueText').innerHTML = "Channeling real-world gold... Strike incoming!";
        document.getElementById('btnStrike').style.display = "none";
        
        // Send the payment to Code.gs to update the spreadsheet
        google.script.run
          .withSuccessHandler(handleStrikeResult)
          .processChannelStrike(paymentAmount, 'fact_venmo', 'sarah');
      }

      function handleStrikeResult(response) {
        if(response.event === "BOSS_DEFEATED") {
          document.getElementById('dialogueText').innerHTML = "CRITICAL HIT! The Vanguard shatters.<br>Zone Unlocked: " + response.zone_unlocked;
        } else {
          document.getElementById('dialogueText').innerHTML = `Direct hit! The Vanguard reels.<br>Remaining HP: $${response.new_hp}`;
          document.getElementById('btnStrike').style.display = "inline-block";
        }
      }
    </script>
